import { useRef, useState, useMemo, useEffect } from 'react'
import { Canvas, useFrame } from '@react-three/fiber'
import { OrbitControls, PerspectiveCamera } from '@react-three/drei'
import * as THREE from 'three'

interface ParticleSceneProps {
  mode: 'idle' | 'gesture' | 'mouse' | 'drawing' | 'result'
  handPosition: { x: number; y: number } | null
  stickLevel: string | null
  onMouseLongPress?: () =&gt; void
  onMouseRelease?: () =&gt; void
}

// 背景粒子系统
const BackgroundParticles = ({ 
  handPosition, 
  mode,
  stickLevel 
}: { 
  handPosition: { x: number; y: number } | null
  mode: string
  stickLevel: string | null
}) =&gt; {
  const pointsRef = useRef&lt;THREE.Points&gt;(null)
  const [particles] = useState(() =&gt; {
    const count = 3000
    const positions = new Float32Array(count * 3)
    const colors = new Float32Array(count * 3)
    
    for (let i = 0; i &lt; count; i++) {
      positions[i * 3] = (Math.random() - 0.5) * 20
      positions[i * 3 + 1] = (Math.random() - 0.5) * 20
      positions[i * 3 + 2] = (Math.random() - 0.5) * 20
      
      // 默认金色调
      colors[i * 3] = 0.9 + Math.random() * 0.1
      colors[i * 3 + 1] = 0.8 + Math.random() * 0.1
      colors[i * 3 + 2] = 0.4 + Math.random() * 0.1
    }
    return { count, positions, colors }
  })

  // 根据签文等级改变粒子颜色
  const getLevelColor = (level: string | null) =&gt; {
    switch (level) {
      case '上上签':
      case '上签':
        return { r: 1, g: 0.85, b: 0.2 } // 金色
      case '中签':
        return { r: 0.9, g: 0.9, b: 0.9 } // 白色
      case '下签':
      case '下下签':
        return { r: 0.6, g: 0.6, b: 0.6 } // 灰色
      default:
        return { r: 0.95, g: 0.85, b: 0.5 } // 默认金色
    }
  }

  useFrame((state) =&gt; {
    if (!pointsRef.current) return
    
    const positions = pointsRef.current.geometry.attributes.position.array as Float32Array
    const colors = pointsRef.current.geometry.attributes.color.array as Float32Array
    const time = state.clock.getElapsedTime()
    
    const levelColor = getLevelColor(stickLevel)
    
    for (let i = 0; i &lt; particles.count; i++) {
      const i3 = i * 3
      
      // 基础漂浮动画
      positions[i3 + 1] += Math.sin(time + i) * 0.002
      
      // 响应手势
      if (handPosition &amp;&amp; (mode === 'idle' || mode === 'gesture' || mode === 'mouse')) {
        const px = positions[i3]
        const py = positions[i3 + 1]
        
        // 计算粒子到手的距离
        const dx = handPosition.x * 10 - px
        const dy = handPosition.y * 10 - py
        const dist = Math.sqrt(dx * dx + dy * dy)
        
        if (dist &lt; 5) {
          // 推开效果
          const force = (5 - dist) / 5
          positions[i3] -= dx * force * 0.02
          positions[i3 + 1] -= dy * force * 0.02
        }
      }
      
      // 更新颜色
      colors[i3] = levelColor.r * (0.8 + Math.random() * 0.2)
      colors[i3 + 1] = levelColor.g * (0.8 + Math.random() * 0.2)
      colors[i3 + 2] = levelColor.b * (0.8 + Math.random() * 0.2)
    }
    
    pointsRef.current.geometry.attributes.position.needsUpdate = true
    pointsRef.current.geometry.attributes.color.needsUpdate = true
  })

  return (
    &lt;points ref={pointsRef}&gt;
      &lt;bufferGeometry&gt;
        &lt;bufferAttribute
          attach="attributes-position"
          count={particles.count}
          array={particles.positions}
          itemSize={3}
        /&gt;
        &lt;bufferAttribute
          attach="attributes-color"
          count={particles.count}
          array={particles.colors}
          itemSize={3}
        /&gt;
      &lt;/bufferGeometry&gt;
      &lt;pointsMaterial
        size={0.08}
        vertexColors
        transparent
        opacity={0.8}
        blending={THREE.AdditiveBlending}
      /&gt;
    &lt;/points&gt;
  )
}

// 签筒粒子系统
const PotParticles = ({ mode }: { mode: string }) =&gt; {
  const pointsRef = useRef&lt;THREE.Points&gt;(null)
  
  // 创建签筒形状的粒子
  const [particles] = useState(() =&gt; {
    const positions: number[] = []
    const colors: number[] = []
    
    // 签筒主体 - 圆柱形状
    const radius = 1.5
    const height = 3
    const particleCount = 800
    
    for (let i = 0; i &lt; particleCount; i++) {
      const angle = Math.random() * Math.PI * 2
      const r = radius * Math.sqrt(Math.random())
      const h = (Math.random() - 0.5) * height
      
      positions.push(
        Math.cos(angle) * r,
        h,
        Math.sin(angle) * r
      )
      
      // 朱红色
      colors.push(0.8, 0.2, 0.2)
    }
    
    return {
      count: positions.length / 3,
      positions: new Float32Array(positions),
      colors: new Float32Array(colors),
    }
  })

  useFrame((state) =&gt; {
    if (!pointsRef.current) return
    const time = state.clock.getElapsedTime()
    
    // 根据模式晃动签筒
    if (mode === 'gesture' || mode === 'mouse') {
      pointsRef.current.rotation.y = time * 0.5
      pointsRef.current.position.y = Math.sin(time * 3) * 0.1
    }
  })

  return (
    &lt;points ref={pointsRef}&gt;
      &lt;bufferGeometry&gt;
        &lt;bufferAttribute
          attach="attributes-position"
          count={particles.count}
          array={particles.positions}
          itemSize={3}
        /&gt;
        &lt;bufferAttribute
          attach="attributes-color"
          count={particles.count}
          array={particles.colors}
          itemSize={3}
        /&gt;
      &lt;/bufferGeometry&gt;
      &lt;pointsMaterial
        size={0.06}
        vertexColors
        transparent
        opacity={0.9}
      /&gt;
    &lt;/points&gt;
  )
}

// 签条粒子系统
const StickParticles = ({ mode }: { mode: string }) =&gt; {
  const sticks = useMemo(() =&gt; {
    return Array.from({ length: 12 }).map((_, index) =&gt; ({
      id: index,
      angle: (index / 12) * Math.PI * 2,
      heightOffset: Math.random() * 0.5,
    }))
  }, [])

  return (
    &lt;group&gt;
      {sticks.map((stick) =&gt; (
        &lt;StickParticle key={stick.id} {...stick} mode={mode} /&gt;
      ))}
    &lt;/group&gt;
  )
}

const StickParticle = ({ 
  angle, 
  heightOffset, 
  mode 
}: { 
  angle: number
  heightOffset: number
  mode: string
}) =&gt; {
  const pointsRef = useRef&lt;THREE.Points&gt;(null)
  
  const [particles] = useState(() =&gt; {
    const positions: number[] = []
    const colors: number[] = []
    
    // 签条 - 细长条形
    const length = 2.5
    const width = 0.15
    const particleCount = 100
    
    for (let i = 0; i &lt; particleCount; i++) {
      positions.push(
        (Math.random() - 0.5) * width,
        Math.random() * length - length / 2,
        (Math.random() - 0.5) * 0.05
      )
      
      // 浅黄色
      colors.push(0.95, 0.9, 0.7)
    }
    
    return {
      count: positions.length / 3,
      positions: new Float32Array(positions),
      colors: new Float32Array(colors),
    }
  })

  useFrame((state) =&gt; {
    if (!pointsRef.current) return
    
    // 签条位置 - 围绕签筒中心排列
    const radius = 0.8
    pointsRef.current.position.x = Math.cos(angle) * radius
    pointsRef.current.position.z = Math.sin(angle) * radius
    pointsRef.current.position.y = heightOffset
    
    // 签条角度指向中心
    pointsRef.current.rotation.y = angle + Math.PI / 2
  })

  return (
    &lt;points ref={pointsRef}&gt;
      &lt;bufferGeometry&gt;
        &lt;bufferAttribute
          attach="attributes-position"
          count={particles.count}
          array={particles.positions}
          itemSize={3}
        /&gt;
        &lt;bufferAttribute
          attach="attributes-color"
          count={particles.count}
          array={particles.colors}
          itemSize={3}
        /&gt;
      &lt;/bufferGeometry&gt;
      &lt;pointsMaterial
        size={0.04}
        vertexColors
        transparent
        opacity={0.95}
      /&gt;
    &lt;/points&gt;
  )
}

export const ParticleScene = ({
  mode,
  handPosition,
  stickLevel,
  onMouseLongPress,
  onMouseRelease,
}: ParticleSceneProps) =&gt; {
  const [isPressing, setIsPressing] = useState(false)
  const pressTimerRef = useRef&lt;NodeJS.Timeout&gt;()

  const handlePointerDown = () =&gt; {
    if (mode === 'mouse') {
      setIsPressing(true)
      pressTimerRef.current = setTimeout(() =&gt; {
        onMouseLongPress?.()
      }, 500)
    }
  }

  const handlePointerUp = () =&gt; {
    if (pressTimerRef.current) {
      clearTimeout(pressTimerRef.current)
    }
    if (isPressing) {
      setIsPressing(false)
      onMouseRelease?.()
    }
  }

  return (
    &lt;div 
      className="w-full h-full"
      onPointerDown={handlePointerDown}
      onPointerUp={handlePointerUp}
      onPointerLeave={handlePointerUp}
    &gt;
      &lt;Canvas&gt;
        &lt;PerspectiveCamera makeDefault position={[0, 0, 8]} fov={60} /&gt;
        &lt;OrbitControls enableZoom={false} enablePan={false} /&gt;
        
        &lt;color attach="background" args={['#0a0a0f']} /&gt;
        
        &lt;BackgroundParticles 
          handPosition={handPosition} 
          mode={mode}
          stickLevel={stickLevel}
        /&gt;
        
        &lt;group position={[0, -0.5, 0]}&gt;
          &lt;PotParticles mode={mode} /&gt;
          &lt;StickParticles mode={mode} /&gt;
        &lt;/group&gt;
      &lt;/Canvas&gt;
    &lt;/div&gt;
  )
}

export type GuanyinGestureType = 'push_open' | 'wave' | 'pull_in' | 'draw' | null

interface UseGuanyinGestureOptions {
  enabled: boolean
  onGestureDetected?: (gesture: GuanyinGestureType) =&gt; void

}

const DEBUG = true

const log = (...args: any[]) =&gt; {
  if (DEBUG) console.log('[GuanyinGesture]', ...args)
}

export const useGuanyinGesture = ({
  enabled,
  onGestureDetected,
  mode,
}: UseGuanyinGestureOptions) =&gt; {
  const videoRef = useRef&lt;HTMLVideoElement&gt;(null)
  const handsRef = useRef&lt;Hands | null&gt;(null)
  const [cameraReady, setCameraReady] = useState(false)

  const [modelLoaded, setModelLoaded] = useState(false)
  const [handPosition, setHandPosition] = useState&lt;{ x: number; y: number } | null&gt;(null)

  const onGestureDetectedRef = useRef(onGestureDetected)

  const gestureStateRef = useRef({
    // 手掌推开检测

    // 左右挥手检测

  }, [onGestureDetected])

    }

      maxNumHands: 1,
      modelComplexity: 0,

    })

    hands.onResults((results: any) =&gt; {
      if (!isDetectingRef.current) return

      const handCount = results.multiHandLandmarks?.length || 0
      if (handCount === 0) {
        setHandPosition(null)
        return
      }

      const landmarks = results.multiHandLandmarks[0]
      
      // 手腕和中指根部的平均位置
      const wrist = landmarks[0]
      const middleBase = landmarks[9]
      const handX = (wrist.x + middleBase.x) / 2
      const handY = (wrist.y + middleBase.y) / 2
      
      // 手的大小（用于检测推开）
      const thumbTip = landmarks[4]
      const pinkyTip = landmarks[20]
      const handSize = Math.sqrt(
        Math.pow(thumbTip.x - pinkyTip.x, 2) + 
        Math.pow(thumbTip.y - pinkyTip.y, 2)
      )

      setHandPosition({ x: handX - 0.5, y: -handY + 0.5 }) // 调整坐标系

      const now = Date.now()
      const state = gestureStateRef.current
      const callback = onGestureDetectedRef.current

      // 根据模式处理手势
      if (mode === 'idle') {
        // 空闲模式：检测推开、挥手、拉近，用于控制粒子
        detectIdleGestures(landmarks, handSize, now, state, callback)
      } else if (mode === 'drawing') {
        // 抽签模式：检测左右挥手来抽签
        detectDrawingGesture(handX, now, state, callback)
      }
    })

    handsRef.current = hands
    setModelLoaded(true)
    log('MediaPipe Hands for Guanyin initialized successfully')

    return () =&gt; {
      log('Cleaning up MediaPipe Hands')
      hands.close()
    }
  }, [enabled, mode])

  // 检测空闲模式手势
  const detectIdleGestures = (
    landmarks: any[],
    handSize: number,
    now: number,
    state: any,
    callback: ((gesture: GuanyinGestureType) =&gt; void) | undefined
  ) =&gt; {
    const wrist = landmarks[0]
    const middleBase = landmarks[9]
    const handX = (wrist.x + middleBase.x) / 2

    // 1. 手掌推开检测（手变大）
    if (!state.pushOpenInitialized) {
      state.lastHandSize = handSize
      state.pushOpenInitialized = true
      return
    }

    const sizeDelta = handSize - state.lastHandSize
    if (sizeDelta &gt; 0.05) {
      log('Push open gesture detected')
      callback?.('push_open')
    }
    state.lastHandSize = handSize

    // 2. 左右挥手检测
    if (!state.waveInitialized) {
      state.lastX = handX
      state.waveInitialized = true
      return
    }

    const deltaX = handX - state.lastX
    const absDeltaX = Math.abs(deltaX)

    if (absDeltaX &gt; 0.03) {
      const newDirection = deltaX &gt; 0 ? 1 : -1
      if (state.waveDirection !== 0 &amp;&amp; state.waveDirection !== newDirection) {
        state.waveCount++
        log('Wave detected, count:', state.waveCount)
        callback?.('wave')
        state.lastWaveTime = now
      }
      state.waveDirection = newDirection
    }
    state.lastX = handX

    // 3. 手往前拉检测（可以用手的Y轴变化模拟）
    const handY = (wrist.y + middleBase.y) / 2
    if (!state.pullInInitialized) {
      state.lastZ = handY
      state.pullInInitialized = true
      return
    }

    const zDelta = state.lastZ - handY // 手向上移动视为拉近
    if (zDelta &gt; 0.08) {
      log('Pull in gesture detected')
      callback?.('pull_in')
    }
    state.lastZ = handY
  }

  // 检测抽签手势
  const detectDrawingGesture = (
    handX: number,
    now: number,
    state: any,
    callback: ((gesture: GuanyinGestureType) =&gt; void) | undefined
  ) =&gt; {
    if (!state.waveInitialized) {
      state.lastX = handX
      state.waveCount = 0
      state.waveInitialized = true
      return
    }

    const deltaX = handX - state.lastX
    const absDeltaX = Math.abs(deltaX)

    if (absDeltaX &gt; 0.02) {
      const newDirection = deltaX &gt; 0 ? 1 : -1
      if (state.waveDirection !== 0 &amp;&amp; state.waveDirection !== newDirection) {
        state.waveCount++
        log('Draw wave detected, count:', state.waveCount)
        if (state.waveCount &gt;= 3 &amp;&amp; now - state.lastWaveTime &gt; 500) {
          log('===== DRAW GESTURE TRIGGERED! =====')
          callback?.('draw')
          state.waveCount = 0
        }
        state.lastWaveTime = now
      }
      state.waveDirection = newDirection
    }
    state.lastX = handX
  }

  const startCamera = useCallback(async () =&gt; {
    log('startCamera called')

    if (!videoRef.current) {
      throw new Error('video element not found')
    }

    try {
      log('Requesting user media...')
      const stream = await navigator.mediaDevices.getUserMedia({
        video: {
          facingMode: 'user',
          width: { ideal: 640 },
          height: { ideal: 480 },
          frameRate: { ideal: 30, max: 60 },
        },
      })

      const video = videoRef.current
      video.srcObject = stream

      await new Promise&lt;void&gt;((resolve, reject) =&gt; {
        const onLoadedMetadata = () =&gt; resolve()
        const onError = () =&gt; reject(new Error('Video error'))
        video.onloadedmetadata = onLoadedMetadata
        video.onerror = onError
        if (video.readyState &gt;= 1) resolve()
      })

      await video.play()
      setCameraReady(true)
    } catch (error) {
      logError('Camera error:', error)
      throw error
    }
  }, [])

  const stopCamera = useCallback(() =&gt; {
    log('Stopping camera...')
    if (videoRef.current?.srcObject) {
      const stream = videoRef.current.srcObject as MediaStream
      stream.getTracks().forEach(track =&gt; track.stop())
      videoRef.current.srcObject = null
    }
    setCameraReady(false)
  }, [])

  const startDetection = useCallback(async () =&gt; {
    if (!modelLoaded || !cameraReady) return

    log('Setting isDetecting = true')
    setIsDetecting(true)
    isDetectingRef.current = true

    // 重置状态
    gestureStateRef.current = {
      lastHandSize: 0,
      pushOpenInitialized: false,
      lastX: 0,
      waveDirection: 0,
      waveCount: 0,
      lastWaveTime: 0,
      waveInitialized: false,
      lastZ: 0,
      pullInInitialized: false,
    }
  }, [modelLoaded, cameraReady])

  const stopDetection = useCallback(() =&gt; {
    log('stopDetection called')
    setIsDetecting(false)
    isDetectingRef.current = false
  }, [])

  // 自动启动检测
  useEffect(() =&gt; {
    if (cameraReady &amp;&amp; modelLoaded &amp;&amp; enabled) {
      const timer = setTimeout(() =&gt; startDetection(), 500)
      return () =&gt; clearTimeout(timer)
    }
  }, [cameraReady, modelLoaded, enabled, startDetection])

  return {
    videoRef,
    handPosition,
    cameraReady,
    isDetecting,
    modelLoaded,
    startCamera,
    stopCamera,
    startDetection,
    stopDetection,
  }
}

    SaveRecordRequest, SaveRecordResponse,
    GuanyinStick

router = APIRouter(prefix="/api/guanyin", tags=["guanyin"])

@router.post("/draw", response_model=DrawResponse)
async def draw_endpoint(request: DrawRequest):
    """抽取一支签"""
    stick = draw_stick()
    return DrawResponse(stick=stick)

@router.post("/interpret", response_model=InterpretResponse)
async def interpret_endpoint(request: InterpretRequest):
    """AI初次解读签文"""
    try:
        interpretation = await ai_agent.interpret_stick(request.question, request.stick)

    except Exception as e:
